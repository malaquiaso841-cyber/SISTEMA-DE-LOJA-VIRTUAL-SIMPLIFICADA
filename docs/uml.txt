```mermaid
    class ItemCarrinho {
        +Produto produto
        +int quantidade
        +float preco_unitario

        +from_dict(dados)

    class Carrinho {
        +Cliente cliente
        +List~ItemCarrinho~ itens

        +int quantidade_total_itens

        +float total

    }

    class Cupom {
        +String codigo
        +String tipo
        +float valor_margem
        +datetime data_validade
        +int uso_maximo
        +List~String~ categorias_elegiveis
        +int usos_realizados
        +validar(categoria)
        +calcular_desconto(subtotal, categoria)
        +registrar_uso(categoria)
        +to_dict()
        +from_dict(dados)
    }

    class Frete {
        +calcular_valor(endereco)
        +calcular_prazo(endereco)
        +calcular(endereco)
    }

    class ItemPedido {
        +String sku
        +int quantidade
        +float preco_unitario
        +bool digital
        +String categoria
        +subtotal
        +to_dict()
        +from_dict(dados)

        +datetime data
        +String forma
        +float valor
        +to_dict()
        +from_dict(dados)
    }

    class Pedido {
        +String id
        +Cliente cliente
        +Endereco endereco_entrega
        +List~ItemPedido~ itens
        +float valor_frete
        +float valor_desconto
        +int prazo_entrega_dias
        +String status
        +String codigo_rastreio
        +datetime data_criacao
        +datetime data_entrega
        +List~Pagamento~ pagamentos
        +calcular_total_pedido()

        +registrar_pagamento(pagamento, produtos)
        +alterar_status(novo_status)
        +cancelar_pedido(janela_horas, produtos)
        +gerar_resumo_nota()

        +from_dict(dados)
        +__str__()

    class Cadastros {
        <<module>>
        +adicionar_produto(produto)
        +adicionar_cliente(cliente)

        +atualizar_cliente(id_cliente, ...)
        +alterar_status_produto(sku, ativo)
        +remover_produto(sku)
        +remover_cliente(id_cliente)
        +criar_cupom(codigo, tipo, valor_margem, uso_maximo)
    }

    class OperacoesPedido {
        <<module>>
        +registrar_pagamento_pedido(pedido_id, pagamento)
        +cancelar_pedido(pedido_id, janela_horas)
        +alterar_status_pedido(pedido_id, novo_status)
    }

    class Relatorios {
        <<module>>

        +gerar_relatorio_pedidos()

        +gerar_relatorio_faturamento_por_periodo(data_inicio, data_fim)
        +gerar_relatorio_top_produtos_e_ticket_medio(limite)

    class Dados {

        +carregar_produtos()

        +carregar_pedidos()
        +salvar_pedidos(pedidos)
        +carregar_cupons()
        +salvar_cupons(cupons)
    }

    class Configuracao {
        <<module>>
        +carregar_configuracoes()
    }

        <<module>>
        +menu()
        +criar_pedido()
        +menu_relatorios()

    class LojaGUI {

    Carrinho "1" *-- "0..*" ItemCarrinho : itens
    ItemCarrinho "*" --> "1" Produto : produto
    Carrinho "*" --> "1" Cliente : cliente

    Pedido "1" *-- "1..*" ItemPedido : itens
    Pedido "*" --> "1" Cliente : cliente
    Pedido "*" --> "1" Endereco : endereco de entrega
    Pedido "1" *-- "0..*" Pagamento : pagamentos
    Pedido ..> Cupom : valida e registra uso
    Pedido ..> Frete : calcula valor e prazo

    Frete ..> Configuracao : lê regras de frete
    Cadastros ..> Dados : persiste cadastros
    Cadastros ..> Configuracao : lê validade padrão do cupom
    OperacoesPedido ..> Pedido : executa operações
    OperacoesPedido ..> Dados : carrega e salva dados
    Relatorios ..> Dados : consulta pedidos e produtos

    MenuCLI ..> Cadastros
    MenuCLI ..> OperacoesPedido
    MenuCLI ..> Relatorios
    LojaGUI ..> Cadastros
    LojaGUI ..> OperacoesPedido
    LojaGUI ..> Relatorios
    Main ..> MenuCLI
    Main ..> LojaGUI
```

Os blocos `Cadastros`, `OperacoesPedido`, `Relatorios`, `Dados`, `Configuracao`, `MenuCLI` e `Main` estão marcados como módulos porque representam arquivos com funções, não classes Python. `prazo_entrega_dias` pode ficar opcional no UML, pois produtos somente digitais não têm prazo de entrega; ele é `None` nesses pedidos.
