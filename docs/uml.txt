classDiagram
    class Produto {
        +str sku
        +str nome
        +str categoria
        +float preco_unitario
        +int estoque
        +bool ativo
        +bool digital
        +atualizar_estoque(quantidade)
        +to_dict()
        +from_dict(dados)
    }

    class Cliente {
        +str id_cliente
        +str nome
        +str email
        +str cpf
        +List~Endereco~ enderecos
        +to_dict()
        +from_dict(dados)
    }

    class Endereco {
        +str logradouro
        +str numero
        +str bairro
        +str cidade
        +str uf
        +str cep
        +to_dict()
        +from_dict(dados)
    }

    class Carrinho {
        +Cliente cliente
        +List~ItemCarrinho~ itens
        +adicionar_item(produto, quantidade)
        +remover_item(sku)
        +calcular_total()
        +limpar()
    }

    class ItemCarrinho {
        +Produto produto
        +int quantidade
        +float preco_unitario
        +float subtotal
    }

    class Pedido {
        +str id
        +Cliente cliente
        +Endereco endereco_entrega
        +List~ItemPedido~ itens
        +float valor_frete
        +float valor_desconto
        +str status
        +str codigo_rastreio
        +datetime data_criacao
        +datetime data_entrega
        +List~Pagamento~ pagamentos
        +fechar_pedido_com_carrinho(carrinho, frete, cupom)
        +calcular_total_pedido()
        +registrar_pagamento(pagamento)
        +alterar_status(novo_status)
        +cancelar_pedido(janela_horas)
        +gerar_resumo_nota()
        +to_dict()
        +from_dict(dados)
    }

    class ItemPedido {
        +str sku
        +int quantidade
        +float preco_unitario
        +bool digital
        +float subtotal
        +to_dict()
        +from_dict(dados)
    }

    class Pagamento {
        +datetime data
        +str forma
        +float valor
        +to_dict()
        +from_dict(dados)
    }

    class Cupom {
        +str codigo
        +str tipo
        +float valor_margem
        +datetime data_validade
        +int uso_maximo
        +validar()
        +calcular_desconto(subtotal)
        +registrar_uso()
        +to_dict()
        +from_dict(dados)
    }

    class Frete {
        +calcular_valor(endereco)
        +calcular_prazo(endereco)
        +calcular(endereco)
    }

    class Dados {
        +carregar_produtos()
        +salvar_produtos(produtos)
        +carregar_clientes()
        +salvar_clientes(clientes)
        +carregar_pedidos()
        +salvar_pedidos(pedidos)
        +carregar_cupons()
        +salvar_cupons(cupons)
    }

    class Relatorios {
        +gerar_relatorio_estoque()
        +gerar_relatorio_pedidos()
        +gerar_relatorio_vendas()
    }

    Cliente "1" *-- "0..*" Endereco : possui
    Carrinho "1" *-- "0..*" ItemCarrinho : contém
    ItemCarrinho "*" --> "1" Produto : referencia
    Pedido "1" *-- "1..*" ItemPedido : contém
    Pedido "*" --> "1" Cliente : pertence a
    Pedido "*" --> "1" Endereco : entrega em
    Pedido "1" *-- "0..*" Pagamento : recebe
    Pedido ..> Carrinho : converte em pedido
    Pedido ..> Cupom : aplica desconto
    Pedido ..> Frete : calcula entrega
    Frete ..> Endereco : consulta UF
    Dados ..> Produto : salva e carrega
    Dados ..> Cliente : salva e carrega
    Dados ..> Pedido : salva e carrega
    Dados ..> Cupom : salva e carrega
